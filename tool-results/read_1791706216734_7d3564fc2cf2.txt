     1→"use client";
     2→/* eslint-disable react-hooks/set-state-in-effect */
     3→
     4→// Vue Saisie — journaux comptables (partie double) + balance + financement.
     5→// Refonte complète avec :
     6→// - Journal achat (Date, N° compte, Libellé, Débit, Crédit)
     7→// - Journal vente (même structure)
     8→// - Balance (somme débit - somme crédit + conclusion)
     9→// - Financement (14 775 000 avec soustractions achat + ajouts vente en cascade)
    10→
    11→import { useState, useEffect, useCallback } from "react";
    12→import { motion, AnimatePresence } from "framer-motion";
    13→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
    14→import { Button } from "@/components/ui/button";
    15→import { Input } from "@/components/ui/input";
    16→import { Label } from "@/components/ui/label";
    17→import { Textarea } from "@/components/ui/textarea";
    18→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    19→import { Badge } from "@/components/ui/badge";
    20→import { HugeiconsIcon } from "@hugeicons/react";
    21→import { Beef, ShoppingCart, Receipt, Package, Plus, AlertTriangle, ChevronDown, ChevronUp, Wifi, Scale, Landmark, Wallet } from "@/lib/icons";
    22→import { formatFCFA, formatDate, severiteColor } from "@/lib/format";
    23→import { addPendingEntry } from "@/lib/offline-db";
    24→import { toast } from "sonner";
    25→
    26→interface JournalEntry {
    27→  id: string;
    28→  date: string;
    29→  numCompte: string;
    30→  libelle: string;
    31→  debit: number;
    32→  credit: number;
    33→}
    34→interface Alerte { type: string; severite: string; message: string; }
    35→interface FinancementMvt {
    36→  date: string;
    37→  libelle: string;
    38→  type: "achat" | "vente";
    39→  montant: number;
    40→  sens: "sortie" | "entree";
    41→  soldeApres: number;
    42→}
    43→type SaisieType = "achat" | "vente" | "depense" | "stock";
    44→
    45→const TYPE_CONFIG: Record<SaisieType, { label: string; icon: typeof Beef; color: string }> = {
    46→  achat: { label: "Journal d'achats", icon: Beef, color: "#10B981" },
    47→  vente: { label: "Journal de ventes", icon: ShoppingCart, color: "#14B8A6" },
    48→  depense: { label: "Journal des dépenses", icon: Receipt, color: "#F59E0B" },
    49→  stock: { label: "Gestion de stock", icon: Package, color: "#3B82F6" },
    50→};
    51→
    52→export function SaisieView() {
    53→  const [activeType, setActiveType] = useState<SaisieType | null>(null);
    54→  const [entries, setEntries] = useState<JournalEntry[]>([]);
    55→  const [alertes, setAlertes] = useState<Alerte[]>([]);
    56→  const [showAlerts, setShowAlerts] = useState(true);
    57→  const [loading, setLoading] = useState(false);
    58→  const [isOnline, setIsOnline] = useState(() => typeof navigator !== "undefined" ? navigator.onLine : true);
    59→  const [totalDebit, setTotalDebit] = useState(0);
    60→  const [totalCredit, setTotalCredit] = useState(0);
    61→  const [solde, setSolde] = useState(0);
    62→  const [conclusion, setConclusion] = useState("");
    63→
    64→  // Données balance
    65→  const [balance, setBalance] = useState<{ numCompte: string; libelle: string; totalDebit: number; totalCredit: number; solde: number; conclusion: string }[]>([]);
    66→  const [showBalance, setShowBalance] = useState(false);
    67→
    68→  // Données financement
    69→  const [financement, setFinancement] = useState<{ financementInitial: number; totalAchats: number; totalVentes: number; soldeRestant: number; mouvements: FinancementMvt[] } | null>(null);
    70→  const [showFinancement, setShowFinancement] = useState(false);
    71→
    72→  const loadJournal = useCallback(async (t: SaisieType) => {
    73→    setLoading(true);
    74→    try {
    75→      const res = await fetch(`/api/saisie?type=${t}`);
    76→      if (!res.ok) return;
    77→      const data = await res.json();
    78→      setEntries(data.entries || []);
    79→      setTotalDebit(data.totalDebit || 0);
    80→      setTotalCredit(data.totalCredit || 0);
    81→      setSolde(data.solde || 0);
    82→      setConclusion(data.conclusion || "");
    83→    } catch { /* hors ligne */ }
    84→    setLoading(false);
    85→  }, []);
    86→
    87→  const loadAlertes = useCallback(async () => {
    88→    try {
    89→      const res = await fetch("/api/saisie?type=alertes");
    90→      if (!res.ok) return;
    91→      const data = await res.json();
    92→      setAlertes(data.alertes || []);
    93→    } catch { /* ignore */ }
    94→  }, []);
    95→
    96→  const loadBalance = useCallback(async () => {
    97→    try {
    98→      const res = await fetch("/api/saisie?type=balance");
    99→      if (!res.ok) return;
   100→      const data = await res.json();
   101→      setBalance(data.balance || []);
   102→    } catch { /* ignore */ }
   103→  }, []);
   104→
   105→  const loadFinancement = useCallback(async () => {
   106→    try {
   107→      const res = await fetch("/api/saisie?type=financement");
   108→      if (!res.ok) return;
   109→      const data = await res.json();
   110→      setFinancement(data);
   111→    } catch { /* ignore */ }
   112→  }, []);
   113→
   114→  useEffect(() => {
   115→    const handleOnlineChange = () => setIsOnline(navigator.onLine);
   116→    window.addEventListener("online", handleOnlineChange);
   117→    window.addEventListener("offline", handleOnlineChange);
   118→    loadAlertes();
   119→    return () => {
   120→      window.removeEventListener("online", handleOnlineChange);
   121→      window.removeEventListener("offline", handleOnlineChange);
   122→    };
   123→  }, []);
   124→
   125→  useEffect(() => {
   126→    if (activeType) {
   127→      loadJournal(activeType);
   128→    }
   129→  }, [activeType]);
   130→
   131→  const handleSave = async (type: SaisieType, data: Record<string, unknown>) => {
   132→    if (!navigator.onLine) {
   133→      await addPendingEntry(type, data);
   134→      toast.success("Saisie enregistrée hors ligne");
   135→      return;
   136→    }
   137→    try {
   138→      const res = await fetch("/api/saisie", {
   139→        method: "POST",
   140→        headers: { "Content-Type": "application/json" },
   141→        body: JSON.stringify({ type, ...data }),
   142→      });
   143→      if (!res.ok) {
   144→        const err = await res.json();
   145→        toast.error(err.error || "Erreur");
   146→        return;
   147→      }
   148→      toast.success("Saisie enregistrée");
   149→      if (activeType) loadJournal(activeType);
   150→      loadAlertes();
   151→      loadBalance();
   152→      loadFinancement();
   153→    } catch {
   154→      await addPendingEntry(type, data);
   155→      toast.warning("Connexion perdue — sauvegardé hors ligne");
   156→    }
   157→  };
   158→
   159→  return (
   160→    <div className="space-y-4">
   161→      {/* Zone d'alertes */}
   162→      {alertes.length > 0 && (
   163→        <Card className="border-amber-200">
   164→          <button onClick={() => setShowAlerts((v) => !v)} className="w-full flex items-center gap-2 p-3">
   165→            <HugeiconsIcon icon={AlertTriangle} size={16} className="text-amber-600" />
   166→            <span className="text-xs font-semibold text-amber-900 flex-1 text-left">Zone d'alertes ({alertes.length})</span>
   167→            {showAlerts ? <HugeiconsIcon icon={ChevronUp} size={16} className="text-amber-600" /> : <HugeiconsIcon icon={ChevronDown} size={16} className="text-amber-600" />}
   168→          </button>
   169→          {showAlerts && (
   170→            <div className="px-3 pb-3 space-y-1.5">
   171→              {alertes.map((a, i) => (
   172→                <div key={i} className={`flex items-center gap-2 rounded-md border px-2.5 py-1.5 text-xs ${severiteColor(a.severite)}`}>
   173→                  <span className="font-medium flex-1">{a.message}</span>
   174→                  <span className="text-[0.55rem] uppercase font-bold shrink-0">{a.type}</span>
   175→                </div>
   176→              ))}
   177→            </div>
   178→          )}
   179→        </Card>
   180→      )}
   181→
   182→      {/* Sélecteur de type */}
   183→      {!activeType && (
   184→        <div className="space-y-3">
   185→          <div className="text-center">
   186→            <h2 className="text-lg font-bold text-foreground">Que voulez-vous saisir ?</h2>
   187→            <p className="text-xs text-muted-foreground mt-0.5">Choisissez un type d'opération</p>
   188→          </div>
   189→          <div className="grid grid-cols-2 gap-3">
   190→            {(Object.keys(TYPE_CONFIG) as SaisieType[]).map((t) => {
   191→              const cfg = TYPE_CONFIG[t];
   192→              return (
   193→                <motion.button key={t} whileTap={{ scale: 0.95 }} onClick={() => setActiveType(t)}
   194→                  className="flex flex-col items-center gap-2 rounded-xl border-2 p-5 transition-colors hover:bg-muted/40"
   195→                  style={{ borderColor: cfg.color + "40" }}>
   196→                  <div className="h-12 w-12 rounded-full flex items-center justify-center" style={{ background: cfg.color + "15" }}>
   197→                    <HugeiconsIcon icon={cfg.icon} size={24} style={{ color: cfg.color }} />
   198→                  </div>
   199→                  <span className="text-sm font-semibold text-foreground">{cfg.label}</span>
   200→                </motion.button>
   201→              );
   202→            })}
   203→          </div>
   204→
   205→          {/* Accès Balance et Financement */}
   206→          <div className="grid grid-cols-2 gap-3 pt-2">
   207→            <button onClick={() => { setShowBalance(true); loadBalance(); }}
   208→              className="flex items-center gap-2 rounded-lg border border-border bg-white p-3 text-xs font-medium hover:bg-muted/40">
   209→              <HugeiconsIcon icon={Scale} size={18} className="text-primary" /> Balance comptable
   210→            </button>
   211→            <button onClick={() => { setShowFinancement(true); loadFinancement(); }}
   212→              className="flex items-center gap-2 rounded-lg border border-border bg-white p-3 text-xs font-medium hover:bg-muted/40">
   213→              <HugeiconsIcon icon={Landmark} size={18} className="text-primary" /> Suivi financement
   214→            </button>
   215→          </div>
   216→        </div>
   217→      )}
   218→
   219→      {/* Modal Balance */}
   220→      {showBalance && (
   221→        <Card>
   222→          <CardHeader><CardTitle className="text-sm flex items-center gap-2"><HugeiconsIcon icon={Scale} size={16} className="text-primary" /> Balance comptable</CardTitle></CardHeader>
   223→          <CardContent>
   224→            <div className="overflow-x-auto">
   225→              <table className="w-full text-xs">
   226→                <thead>
   227→                  <tr className="border-b-2 border-border text-left">
   228→                    <th className="p-2" rowSpan={2}>Compte</th>
   229→                    <th className="p-2" rowSpan={2}>Libellé</th>
   230→                    <th className="p-2 text-center border-l border-border" colSpan={2}>Solde N-1</th>
   231→                    <th className="p-2 text-center border-l border-border" colSpan={2}>Mouvements N</th>
   232→                    <th className="p-2 text-center border-l border-border" colSpan={2}>Solde N</th>
   233→                  </tr>
   234→                  <tr className="border-b border-border text-left text-[0.6rem]">
   235→                    <th className="p-2 text-right border-l border-border">Débit</th>
   236→                    <th className="p-2 text-right">Crédit</th>
   237→                    <th className="p-2 text-right border-l border-border">Débit</th>
   238→                    <th className="p-2 text-right">Crédit</th>
   239→                    <th className="p-2 text-right border-l border-border">Débit</th>
   240→                    <th className="p-2 text-right">Crédit</th>
   241→                  </tr>
   242→                </thead>
   243→                <tbody>
   244→                  {balance.map((c, i) => {
   245→                    const soldeDebit = c.solde > 0 ? Math.abs(c.solde) : 0;
   246→                    const soldeCredit = c.solde < 0 ? Math.abs(c.solde) : 0;
   247→                    return (
   248→                      <tr key={i} className="border-b border-border/50 hover:bg-muted/30">
   249→                        <td className="p-2 font-mono font-bold">{c.numCompte}</td>
   250→                        <td className="p-2 text-foreground">{c.libelle}</td>
   251→                        <td className="p-2 text-right tabular-nums text-muted-foreground">—</td>
   252→                        <td className="p-2 text-right tabular-nums text-muted-foreground">—</td>
   253→                        <td className="p-2 text-right tabular-nums text-red-600">{c.totalDebit > 0 ? formatFCFA(c.totalDebit, false) : ""}</td>
   254→                        <td className="p-2 text-right tabular-nums text-emerald-600">{c.totalCredit > 0 ? formatFCFA(c.totalCredit, false) : ""}</td>
   255→                        <td className="p-2 text-right tabular-nums font-bold text-red-600">{soldeDebit > 0 ? formatFCFA(soldeDebit, false) : ""}</td>
   256→                        <td className="p-2 text-right tabular-nums font-bold text-emerald-600">{soldeCredit > 0 ? formatFCFA(soldeCredit, false) : ""}</td>
   257→                      </tr>
   258→                    );
   259→                  })}
   260→                </tbody>
   261→                <tfoot>
   262→                  <tr className="border-t-2 border-border font-bold bg-muted/30">
   263→                    <td className="p-2" colSpan={2}>Total général</td>
   264→                    <td className="p-2 text-right tabular-nums text-muted-foreground">—</td>
   265→                    <td className="p-2 text-right tabular-nums text-muted-foreground">—</td>
   266→                    <td className="p-2 text-right tabular-nums text-red-600">{formatFCFA(balance.reduce((s, c) => s + c.totalDebit, 0), false)}</td>
   267→                    <td className="p-2 text-right tabular-nums text-emerald-600">{formatFCFA(balance.reduce((s, c) => s + c.totalCredit, 0), false)}</td>
   268→                    <td className="p-2 text-right tabular-nums text-red-600">{formatFCFA(Math.max(0, balance.reduce((s, c) => s + c.solde, 0)), false)}</td>
   269→                    <td className="p-2 text-right tabular-nums text-emerald-600">{formatFCFA(Math.max(0, -balance.reduce((s, c) => s + c.solde, 0)), false)}</td>
   270→                  </tr>
   271→                </tfoot>
   272→              </table>
   273→            </div>
   274→            <div className="mt-3 flex items-center justify-between rounded-lg border border-border px-3 py-2">
   275→              <span className="text-xs text-muted-foreground">Solde général (Débit − Crédit)</span>
   276→              <div className="flex items-center gap-2">
   277→                <span className={`text-sm font-bold tabular-nums ${balance.reduce((s, c) => s + c.solde, 0) > 0 ? "text-red-600" : "text-emerald-600"}`}>
   278→                  {formatFCFA(Math.abs(balance.reduce((s, c) => s + c.solde, 0)), false)}
   279→                </span>
   280→                <Badge variant="outline" className={balance.reduce((s, c) => s + c.solde, 0) > 0 ? "text-red-700 border-red-300 bg-red-50" : "text-emerald-700 border-emerald-300 bg-emerald-50"}>
   281→                  {balance.reduce((s, c) => s + c.solde, 0) > 0 ? "Solde débitaire" : "Solde créditaire"}
   282→                </Badge>
   283→              </div>
   284→            </div>
   285→            <button onClick={() => setShowBalance(false)} className="mt-3 text-xs text-muted-foreground hover:text-foreground">← Fermer</button>
   286→          </CardContent>
   287→        </Card>
   288→      )}
   289→
   290→      {/* Modal Financement en cascade */}
   291→      {showFinancement && financement && (
   292→        <Card>
   293→          <CardHeader>
   294→            <CardTitle className="text-sm flex items-center gap-2"><HugeiconsIcon icon={Landmark} size={16} className="text-primary" /> Suivi financement</CardTitle>
   295→            <CardDescription className="text-xs">Capital initial : {formatFCFA(financement.financementInitial)}</CardDescription>
   296→          </CardHeader>
   297→          <CardContent className="space-y-3">
   298→            {/* Résumé */}
   299→            <div className="grid grid-cols-3 gap-2">
   300→              <div className="rounded-lg bg-red-50 border border-red-200 p-2 text-center">
   301→                <p className="text-[0.6rem] uppercase text-red-700 font-medium">Achats (sorties)</p>
   302→                <p className="text-sm font-bold text-red-700 tabular-nums">{formatFCFA(financement.totalAchats, false)}</p>
   303→              </div>
   304→              <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-2 text-center">
   305→                <p className="text-[0.6rem] uppercase text-emerald-700 font-medium">Ventes (entrées)</p>
   306→                <p className="text-sm font-bold text-emerald-700 tabular-nums">{formatFCFA(financement.totalVentes, false)}</p>
   307→              </div>
   308→              <div className="rounded-lg bg-primary/5 border border-primary/20 p-2 text-center">
   309→                <p className="text-[0.6rem] uppercase text-primary font-medium">Solde restant</p>
   310→                <p className="text-sm font-bold text-primary tabular-nums">{formatFCFA(financement.soldeRestant, false)}</p>
   311→              </div>
   312→            </div>
   313→
   314→            {/* Mouvements en cascade */}
   315→            <div className="space-y-0">
   316→              <AnimatePresence mode="popLayout">
   317→                {financement.mouvements.map((m, i) => (
   318→                  <motion.div
   319→                    key={`${m.date}-${i}`}
   320→                    layout
   321→                    initial={{ opacity: 0, x: -30, height: 0 }}
   322→                    animate={{ opacity: 1, x: 0, height: "auto", marginLeft: Math.max(0, 16 - i * 3) }}
   323→                    exit={{ opacity: 0, x: -30, height: 0 }}
   324→                    transition={{ delay: i * 0.06, duration: 0.3 }}
   325→                    className={`relative overflow-hidden rounded-lg border p-3 ${m.sens === "sortie" ? "border-red-200 bg-red-50/30" : "border-emerald-200 bg-emerald-50/30"}`}
   326→                  >
   327→                    <div className={`absolute left-0 top-0 bottom-0 w-1 ${m.sens === "sortie" ? "bg-red-500" : "bg-emerald-500"}`} />
   328→                    <div className="flex items-center justify-between gap-2 pl-2">
   329→                      <div className="min-w-0 flex-1">
   330→                        <p className="text-sm font-medium text-foreground truncate">{m.libelle}</p>
   331→                        <p className="text-[0.6rem] text-muted-foreground">{m.date} · {m.sens === "sortie" ? "Achat (sortie)" : "Vente (entrée)"}</p>
   332→                      </div>
   333→                      <div className="text-right shrink-0">
   334→                        <p className={`text-sm font-bold tabular-nums ${m.sens === "sortie" ? "text-red-600" : "text-emerald-600"}`}>
   335→                          {m.sens === "sortie" ? "−" : "+"} {formatFCFA(m.montant, false)}
   336→                        </p>
   337→                        <p className="text-[0.55rem] text-muted-foreground">Solde : {formatFCFA(m.soldeApres, false)}</p>
   338→                      </div>
   339→                    </div>
   340→                  </motion.div>
   341→                ))}
   342→              </AnimatePresence>
   343→              {financement.mouvements.length === 0 && (
   344→                <p className="text-center text-xs text-muted-foreground py-4">Aucun mouvement</p>
   345→              )}
   346→            </div>
   347→
   348→            <button onClick={() => setShowFinancement(false)} className="text-xs text-muted-foreground hover:text-foreground">← Fermer</button>
   349→          </CardContent>
   350→        </Card>
   351→      )}
   352→
   353→      {/* Formulaire + Journal quand un type est sélectionné */}
   354→      {activeType && (
   355→        <div className="space-y-4">
   356→          <div className="flex items-center gap-3">
   357→            <button onClick={() => { setActiveType(null); setEntries([]); }} className="flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground">
   358→              <HugeiconsIcon icon={ChevronDown} size={14} className="rotate-90" /> Retour
   359→            </button>
   360→            <div className="flex items-center gap-2 flex-1">
   361→              <div className="h-8 w-8 rounded-full flex items-center justify-center" style={{ background: TYPE_CONFIG[activeType].color + "15" }}>
   362→                <HugeiconsIcon icon={TYPE_CONFIG[activeType].icon} size={16} style={{ color: TYPE_CONFIG[activeType].color }} />
   363→              </div>
   364→              <h2 className="text-base font-bold text-foreground">{TYPE_CONFIG[activeType].label}</h2>
   365→            </div>
   366→            <div className={`flex items-center gap-1 text-[0.6rem] ${isOnline ? "text-emerald-600" : "text-amber-600"}`}>
   367→              <HugeiconsIcon icon={Wifi} size={12} />{isOnline ? "En ligne" : "Hors ligne"}
   368→            </div>
   369→          </div>
   370→
   371→          {/* Formulaire */}
   372→          <Card><CardContent className="p-4">
   373→            {activeType === "achat" && <FormAchat onSave={handleSave} />}
   374→            {activeType === "vente" && <FormVente onSave={handleSave} />}
   375→            {activeType === "depense" && <FormDepense onSave={handleSave} />}
   376→            {activeType === "stock" && <FormStock onSave={handleSave} />}
   377→          </CardContent></Card>
   378→
   379→          {/* Journal comptable — tableau en cascade */}
   380→          <div>
   381→            <div className="flex items-center justify-between mb-2">
   382→              <h3 className="text-sm font-semibold text-foreground">{TYPE_CONFIG[activeType].label}</h3>
   383→              {entries.length > 0 && (
   384→                <div className="text-right">
   385→                  <span className="text-[0.6rem] text-muted-foreground">Solde : </span>
   386→                  <span className={`text-xs font-bold tabular-nums ${solde > 0 ? "text-red-600" : "text-emerald-600"}`}>
   387→                    {formatFCFA(Math.abs(solde), false)} · {conclusion}
   388→                  </span>
   389→                </div>
   390→              )}
   391→            </div>
   392→
   393→            {/* En-tête du tableau */}
   394→            <div className="grid grid-cols-[60px_60px_1fr_80px_80px] gap-2 px-3 py-1.5 border-b border-border text-[0.6rem] font-semibold uppercase tracking-wider text-muted-foreground">
   395→              <span>Date</span>
   396→              <span>N° Compte</span>
   397→              <span>Libellé</span>
   398→              <span className="text-right">Montant Débit</span>
   399→              <span className="text-right">Montant Crédit</span>
   400→            </div>
   401→
   402→            {/* Lignes du journal en cascade */}
   403→            {loading ? (
   404→              <p className="text-center text-xs text-muted-foreground py-4">Chargement...</p>
   405→            ) : entries.length === 0 ? (
   406→              <p className="text-center text-xs text-muted-foreground py-4">Aucune entrée</p>
   407→            ) : (
   408→              <div className="overflow-hidden">
   409→                <AnimatePresence mode="popLayout">
   410→                  {entries.map((e, i) => (
   411→                    <motion.div
   412→                      key={e.id}
   413→                      layout
   414→                      initial={{ opacity: 0, height: 0 }}
   415→                      animate={{ opacity: 1, height: "auto" }}
   416→                      exit={{ opacity: 0, height: 0 }}
   417→                      transition={{ delay: i * 0.04, duration: 0.2 }}
   418→                      className={`grid grid-cols-[60px_60px_1fr_80px_80px] gap-2 items-center px-3 py-1.5 border-b border-border/40 text-xs ${e.debit > 0 ? "bg-red-50/30" : "bg-emerald-50/30"}`}
   419→                    >
   420→                      <span className="text-muted-foreground tabular-nums">{e.date}</span>
   421→                      <span className="font-mono font-bold text-foreground">{e.numCompte}</span>
   422→                      <span className="text-foreground truncate">{e.libelle}</span>
   423→                      <span className="text-right tabular-nums font-medium text-red-600">{e.debit > 0 ? formatFCFA(e.debit, false) : ""}</span>
   424→                      <span className="text-right tabular-nums font-medium text-emerald-600">{e.credit > 0 ? formatFCFA(e.credit, false) : ""}</span>
   425→                    </motion.div>
   426→                  ))}
   427→                </AnimatePresence>
   428→              </div>
   429→            )}
   430→
   431→            {/* Totaux du journal */}
   432→            {entries.length > 0 && (
   433→              <div className="grid grid-cols-[60px_60px_1fr_80px_80px] gap-2 items-center px-3 py-2 bg-muted/40 border border-border rounded-b-lg text-xs font-bold">
   434→                <span></span>
   435→                <span></span>
   436→                <span className="text-muted-foreground">Total</span>
   437→                <span className="text-right tabular-nums text-red-600">{formatFCFA(totalDebit, false)}</span>
   438→                <span className="text-right tabular-nums text-emerald-600">{formatFCFA(totalCredit, false)}</span>
   439→              </div>
   440→            )}
   441→
   442→            {/* Conclusion */}
   443→            {entries.length > 0 && (
   444→              <div className="mt-2 flex items-center justify-between rounded-lg border border-border px-3 py-2">
   445→                <span className="text-xs text-muted-foreground">Solde (Débit − Crédit)</span>
   446→                <div className="flex items-center gap-2">
   447→                  <span className={`text-sm font-bold tabular-nums ${solde > 0 ? "text-red-600" : "text-emerald-600"}`}>
   448→                    {formatFCFA(Math.abs(solde), false)}
   449→                  </span>
   450→                  <Badge variant="outline" className={solde > 0 ? "text-red-700 border-red-300 bg-red-50" : "text-emerald-700 border-emerald-300 bg-emerald-50"}>
   451→                    {conclusion}
   452→                  </Badge>
   453→                </div>
   454→              </div>
   455→            )}
   456→          </div>
   457→        </div>
   458→      )}
   459→    </div>
   460→  );
   461→}
