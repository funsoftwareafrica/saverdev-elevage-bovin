   643→// --- SECTION RENTABILITÉ — waterfall + barre empilée + jauges ---
   644→function RentabiliteSection({
   645→  ca, coutAchat, coutEngraissement, margeTotale, margeParTete,
   646→}: {
   647→  ca: number; coutAchat: number; coutEngraissement: number; margeTotale: number; margeParTete: number;
   648→}) {
   649→  const totalCouts = coutAchat + coutEngraissement;
   650→  const tauxMarge = ca > 0 ? Math.round((margeTotale / ca) * 100) : 0;
   651→  const tauxCoutAchat = ca > 0 ? Math.round((coutAchat / ca) * 100) : 0;
   652→  const tauxCoutEngrais = ca > 0 ? Math.round((coutEngraissement / ca) * 100) : 0;
   653→
   654→  // Données waterfall : CA (start) → -Coût achat → -Coût engrais → Marge (end)
   655→  const waterfallData = [
   656→    { label: "CA", value: ca, type: "total", color: "#10B981" },
   657→    { label: "Coût achat", value: -coutAchat, type: "negative", color: "#EF4444" },
   658→    { label: "Coût engrais.", value: -coutEngraissement, type: "negative", color: "#F59E0B" },
   659→    { label: "Marge", value: margeTotale, type: "total", color: margeTotale >= 0 ? "#14B8A6" : "#EF4444" },
   660→  ];
   661→
   662→  // Calcul des positions cumulées pour le waterfall
   663→  let cumul = 0;
   664→  const bars = waterfallData.map((item) => {
   665→    let base = 0, hauteur = 0;
   666→    if (item.type === "total") {
   667→      base = 0;
   668→      hauteur = item.value;
   669→      cumul = item.value;
   670→    } else {
   671→      base = cumul + item.value; // item.value est négatif
   672→      hauteur = Math.abs(item.value);
   673→      cumul = base;
   674→    }
   675→    return { ...item, base, hauteur };
   676→  });
   677→
   678→  const maxVal = ca;
   679→  const chartH = 200;
   680→
   681→  return (
   682→    <section>
   683→      <SectionTitle icon={TrendingUp} title="Rentabilité" subtitle="Chiffre d'affaires et marges" />
   684→
   685→      {/* Ligne 1 : 4 KPI cards avec mini jauges */}
   686→      <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4 mb-4">
   687→        <RentabiliteKpi
   688→          icon={ShoppingCart} label="Chiffre d'affaires" value={formatFCFAShort(ca)}
   689→          color="#10B981" pct={100} hint="Ventes réalisées" sub={`${tauxCoutAchat + tauxCoutEngrais}% absorbé`}
   690→        />
   691→        <RentabiliteKpi
   692→          icon={Beef} label="Coût d'achat" value={formatFCFAShort(coutAchat)}
   693→          color="#EF4444" pct={tauxCoutAchat} hint="Bétail acheté" sub={`${tauxCoutAchat}% du CA`}
   694→        />
   695→        <RentabiliteKpi
   696→          icon={Salad} label="Coûts d'engraissement" value={formatFCFAShort(coutEngraissement)}
   697→          color="#F59E0B" pct={tauxCoutEngrais} hint="Alimentation + soins" sub={`${tauxCoutEngrais}% du CA`}
   698→        />
   699→        <RentabiliteKpi
   700→          icon={TrendingUp} label="Marge totale" value={formatFCFAShort(margeTotale)}
   701→          color={margeTotale >= 0 ? "#14B8A6" : "#EF4444"} pct={Math.min(100, tauxMarge)}
   702→          hint={`Marge / tête : ${formatFCFA(margeParTete)}`} sub={`${tauxMarge}% du CA`}
   703→        />
   704→      </div>
   705→
   706→      {/* Ligne 2 : waterfall + barre empilée + jauge marge */}
   707→      <div className="grid gap-4 lg:grid-cols-3">
   708→        {/* Waterfall chart (cascade de marge) */}
   709→        <Card className="hover-lift lg:col-span-2">
   710→          <CardHeader className="pb-2">
   711→            <CardTitle className="text-sm flex items-center gap-2"><HugeiconsIcon icon={BarChart3} size={16} className="text-primary" />Cascade de marge</CardTitle>
   712→            <CardDescription className="text-xs">Du CA à la marge — décomposition (FCFA)</CardDescription>
   713→          </CardHeader>
   714→          <CardContent className="pt-2">
   715→            {/* Waterfall HTML/CSS — cascade CA → -coûts → marge */}
   716→            {(() => {
   717→              const containerH = 240; // px
   718→              return (
   719→                <div style={{ height: containerH + 30 }}>
   720→                  <div className="flex items-end justify-around gap-3 h-full pt-4 pb-2">
   721→                    {bars.map((b, i) => {
   722→                      const barH = Math.max(14, Math.round((b.hauteur / maxVal) * containerH));
   723→                      const spacerH = b.type === "total" ? 0 : Math.round((b.base / maxVal) * containerH);
   724→                      return (
   725→                        <div key={i} className="flex-1 max-w-[100px] h-full flex flex-col justify-end items-center">

   937→function HeroKpi({
   938→  icon: Icon, label, value, suffix, formatter, color, trend, trendUp, hint, sparkData,
   939→}: {
   940→  icon: IconSvgElement; label: string; value: number; suffix?: string;
   941→  formatter?: (n: number) => string; color: "emerald" | "teal" | "amber" | "red";
   942→  trend?: string; trendUp?: boolean; hint?: string; sparkData: { v: number }[];
   943→}) {
   944→  const colors: Record<string, { bg: string; text: string; stroke: string; gradient: string; cube: string }> = {
   945→    emerald: { bg: "from-emerald-500/10 to-emerald-500/5", text: "text-emerald-700", stroke: "#10B981", gradient: "gSparkE", cube: "#10B981" },

   947→    amber: { bg: "from-amber-500/10 to-amber-500/5", text: "text-amber-700", stroke: "#F59E0B", gradient: "gSparkA", cube: "#F59E0B" },
   948→    red: { bg: "from-red-500/10 to-red-500/5", text: "text-red-700", stroke: "#EF4444", gradient: "gSparkR", cube: "#EF4444" },
   949→  };
   950→  const c = colors[color];
   951→
   952→  // Parallaxe 3D au survol
   953→  const ref = useRef<HTMLDivElement>(null);
   954→  const mouseX = useMotionValue(0);
   955→  const mouseY = useMotionValue(0);
   956→  const rotateX = useSpring(useTransform(mouseY, [-0.5, 0.5], [12, -12]), { stiffness: 200, damping: 18 });
   957→  const rotateY = useSpring(useTransform(mouseX, [-0.5, 0.5], [-12, 12]), { stiffness: 200, damping: 18 });
   958→
   959→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   960→    const rect = ref.current?.getBoundingClientRect();
   961→    if (!rect) return;
   962→    mouseX.set((e.clientX - rect.left) / rect.width - 0.5);
   963→    mouseY.set((e.clientY - rect.top) / rect.height - 0.5);
   964→  };
   965→  const handleMouseLeave = () => { mouseX.set(0); mouseY.set(0); };
   966→
   967→  return (
   968→    <motion.div
   969→      initial={{ opacity: 0, y: 16 }}
   970→      whileInView={{ opacity: 1, y: 0 }}
   971→      viewport={{ once: true }}
   972→      transition={{ duration: 0.4 }}
   973→      style={{ perspective: "800px" }}
   974→    >
   975→      <motion.div
   976→        ref={ref}
   977→        onMouseMove={handleMouseMove}
   978→        onMouseLeave={handleMouseLeave}
   979→        style={{ rotateX, rotateY, transformStyle: "preserve-3d" }}
   980→      >
   981→        <Card className={`bg-gradient-to-br ${c.bg} border-border hover-lift overflow-hidden relative`}>
   982→          <CardContent className="p-4 sm:p-5">
   983→            <div className="flex items-start justify-between gap-2">

   985→                <p className="text-[0.65rem] uppercase tracking-wider font-medium text-muted-foreground truncate">{label}</p>
   986→                <p className="text-2xl sm:text-3xl font-bold mt-1 text-foreground tabular-nums">
   987→                  <AnimatedCounter value={value} format={formatter} duration={1.4} delay={0.1} />
   988→                  {suffix && <span className="text-base font-medium text-muted-foreground ml-1">{suffix}</span>}
   989→                </p>
   990→                {trend && (
   991→                  <div className="flex items-center gap-1 mt-1 text-[0.7rem] font-medium">
   992→                    {trendUp ? <HugeiconsIcon icon={TrendingUp} size={12} className="text-emerald-600" /> : <HugeiconsIcon icon={Minus} size={12} className="text-muted-foreground" />}
   993→                    <span className={trendUp ? "text-emerald-700" : "text-muted-foreground"}>{trend}</span>
   994→                  </div>
   995→                )}
   996→                {hint && <p className="text-[0.65rem] text-muted-foreground mt-0.5">{hint}</p>}
   997→              </div>
   998→              {/* Cube 3D rotatif avec l'icône */}
   999→              <motion.div
  1000→                className="relative shrink-0"
  1001→                style={{ width: 36, height: 36, transformStyle: "preserve-3d", transform: "translateZ(50px)" }}
  1002→                animate={{ rotateY: [0, 360] }}
  1003→                transition={{ duration: 10, repeat: Infinity, ease: "linear" }}
  1004→              >
  1005→                {/* Face avant */}
  1006→                <div className="absolute inset-0 rounded-lg flex items-center justify-center shadow-md" style={{ background: `linear-gradient(135deg, ${c.cube}, ${c.cube}cc)`, transform: "translateZ(8px)" }}>
  1007→                  <HugeiconsIcon icon={Icon} size={18} className="text-white" />
